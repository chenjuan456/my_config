_zsh_vi_cursor() {
  case $KEYMAP in
    vicmd)      echo -ne '\e[2 q' ;;  # 方块
    viins|main) echo -ne '\e[6 q' ;;  # 竖线
  esac
}
zle -N zle-keymap-select _zsh_vi_cursor
zle -N zle-line-init _zsh_vi_cursor
add-zsh-hook precmd _zsh_vi_cursor

# 退出 zsh 时恢复终端默认光标
_restore_cursor() { echo -ne '\e[0 q' }
add-zsh-hook zshexit _restore_cursor

# 7. Yazi 文件管理器包装函数
# ============================================================
function y() {
	local tmp="$(mktemp -t "yazi-cwd.XXXXXX")"
	yazi "$@" --cwd-file="$tmp"
	if cwd="$(command cat -- "$tmp")" && [ -n "$cwd" ] && [ "$cwd" != "$PWD" ]; then
		builtin cd -- "$cwd"
	fi
	rm -f -- "$tmp"
}
