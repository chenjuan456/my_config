" Enter 处理：补全确认 > 括号展开 (仅 C/C++) > 普通换行
function! s:should_expand_bracket() abort
  if index(['c', 'cpp', 'objc', 'objcpp'], &filetype) < 0
    return 0
  endif
  let line = getline('.')
  let col = col('.') - 1
  if col <= 0 || col >= len(line)
    return 0
  endif
  let before = line[col - 1]
  let after  = line[col]
  return (before == '(' && after == ')')
        \ || (before == '{' && after == '}')
        \ || (before == '[' && after == ']')
endfunction

function! s:cr_handler() abort
  if coc#pum#visible()
    return coc#pum#confirm()
  endif
  if pumvisible()
    return "\<C-y>"
  endif
  if s:should_expand_bracket()
    return "\<CR>\<ESC>=kO"
  endif
  return "\<CR>"
endfunction
inoremap <silent><expr> <CR> <SID>cr_handler()
